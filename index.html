rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ---------- Identité et rôles ---------- function signed() { return request.auth != null && request.auth.token.email_verified == true; } function memberPath() { return /databases/$(database)/documents/membres/$(request.auth.token.email.lower()); } function isMember() { return signed() && exists(memberPath()); } function role() { return get(memberPath()).data.role.lower(); } function canWrite() { return isMember() && role() in ['admin', 'editeur', 'éditeur']; } function isAdmin() { return isMember() && role() == 'admin'; } // ---------- Validation des données ---------- function str(v, max) { return v is string && v.size() <= max; } function isHex(v) { return v is string && v.matches('^#[0-9a-fA-F]{6}$'); } function validPlan(d) { return d.keys().hasOnly(['date', 'net', 'theme', 'tpl', 'done']) && d.keys().hasAll(['date', 'net', 'theme', 'tpl', 'done']) && d.date is string && d.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') && d.net in ['Instagram', 'Story Instagram', 'Facebook', 'LinkedIn'] && str(d.theme, 200) && d.tpl in ['event', 'action', 'quote', 'info'] && d.done is bool; } function validModele(d) { return d.keys().hasOnly(['name', 'data']) && d.keys().hasAll(['name', 'data']) && str(d.name, 80) && str(d.data, 30000); } function validCouleur(d) { return d.keys().hasOnly(['n', 'bg', 'fg']) && d.keys().hasAll(['n', 'bg', 'fg']) && str(d.n, 40) && isHex(d.bg) && isHex(d.fg); } function validLogo(v) { return v is string && v.size() <= 150000 && (v == '' || v.matches('^data:image/(jpeg|png);base64,.*')); } function validConfig(d) { return d.keys().hasOnly(['nom', 'cpt', 'tags', 'sig', 'ville', 'police', 'idpays', 'logo', 'logoN', 'logoB', 'logoR', 'logoQ']) && str(d.nom, 100) && str(d.cpt, 100) && str(d.tags, 300) && str(d.sig, 300) && str(d.ville, 40) && d.police in ['officielle', 'condensee'] && d.idpays is bool; } function validMarque(d) { return d.keys().hasOnly(['data']) && d.data != '' && validLogo(d.data); } function validPolice(d) { return d.keys().hasOnly(['data', 'type']) && d.data is string && d.data.size() > 100 && d.data.size() <= 700000 && d.type in ['woff2', 'woff', 'ttf', 'otf']; } // ---------- Collections ---------- // Fiche de membre : lisible par la personne concernée et par les admins. // Seuls les admins écrivent, et jamais sur leur propre fiche (évite de se retirer ou de s'enlever le rôle admin). match /membres/{email} { allow read: if isAdmin() || (signed() && request.auth.token.email.lower() == email); allow create, update: if isAdmin() && request.auth.token.email.lower() != email && request.resource.data.keys().hasOnly(['role']) && request.resource.data.role in ['admin', 'editeur', 'lecteur']; allow delete: if isAdmin() && request.auth.token.email.lower() != email; } match /plan/{id} { allow read: if isMember(); allow create, update: if canWrite() && validPlan(request.resource.data); allow delete: if canWrite(); } match /modeles/{id} { allow read: if isMember(); allow create, update: if canWrite() && validModele(request.resource.data); allow delete: if canWrite(); } match /couleurs/{id} { allow read: if isMember(); allow create, update: if isAdmin() && validCouleur(request.resource.data); allow delete: if isAdmin(); } match /config/antenne { allow read: if isMember(); allow create, update: if isAdmin() && validConfig(request.resource.data); } // Logos et bougies officiels (une fiche par version) : lecture pour les membres, écriture admin match /marques/{id} { allow read: if isMember(); allow create, update: if isAdmin() && id in ['c', 'n', 'b', 'k', 'y', 'w', 'r', 'q', 's', 't'] && validMarque(request.resource.data); allow delete: if isAdmin(); } // Polices officielles importées (Condensed n°20, Bold n°2, Médium, Light) match /polices/{id} { allow read: if isMember(); allow create, update: if isAdmin() && id in ['cond', 'bold', 'med', 'light'] && validPolice(request.resource.data); allow delete: if isAdmin(); } // Tout le reste est interdit match /{document=**} { allow read, write: if false; } } }
